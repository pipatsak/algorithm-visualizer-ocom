Trial Division O(√N)
vector<pair<long long, int>> factorize(long long n) {
  vector<pair<long long, int>> factors;
  if (n % 2 == 0) { // จัดการเลขคู่ (2)
    int cnt = 0;
    while (n % 2 == 0) { cnt++; n /= 2; }
    factors.push_back({2, cnt});
  }
  for (long long d = 3; d * d <= n; d += 2) {
    if (n % d == 0) { // พบตัวประกอบคี่
      int cnt = 0;
      while (n % d == 0) { cnt++; n /= d; }
      factors.push_back({d, cnt});
    }
  }
  if (n > 1) factors.push_back({n, 1}); // ตัวประกอบสุดท้าย > √n
  return factors;
}

ทำไมลูปข้ามเลขประกอบได้?

แม้ลูปจะเพิ่ม $d \mathrel{+}= 2$ ผ่านเลขประกอบ เช่น $9, 15$ แต่มันจะหาร $n$ ไม่ลงตัวแน่นอน เพราะตัวประกอบเฉพาะของมัน (เช่น 3) ถูกหารทอนออกจนหมดเกลี้ยงในรอบก่อนหน้าแล้ว

ทำไมต้องมี if (n > 1)?

หาก $n$ มีตัวประกอบเฉพาะที่มากกว่า $\sqrt{n}$ (เช่น $46 = 2 \times 23$) ลูปจะหยุดที่ $d \le \sqrt{46} \approx 6$ ทำให้เหลือค่า $n = 23$ ตกค้าง จึงต้องเก็บตกด้วย if (n > 1) เสมอ

เทคนิค SPF สำหรับหลายคำถาม

หากโจทย์มี $Q = 10^5$ คำถาม ให้ใช้ Sieve สร้างตาราง spf[x] (Smallest Prime Factor) ไว้ก่อน จะช่วยให้แยกตัวประกอบแต่ละคำถามได้เร็วในเวลา $O(\log N)$ แทน $O(\sqrt{N})$
