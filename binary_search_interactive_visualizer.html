binary_search.cpp
C++ Implementation
int binarySearch(const vector<int>& arr, int target) {
  int low = 0, high = (int)arr.size() - 1;
  while (low <= high) {
    int mid = low + (high - low) / 2;
    if (arr[mid] == target) {
      return mid; // พบข้อมูล คืน index
    } else if (arr[mid] < target) {
      low = mid + 1; // ตัดครึ่งซ้าย หาทางขวา
    } else {
      high = mid - 1; // ตัดครึ่งขวา หาทางซ้าย
    }
  }
  return -1; // หมดช่วงค้นหาแล้วไม่พบ
}
แถบสีสว่าง = บรรทัดที่กำลังประมวลผล Safe mid formula (ป้องกัน Overflow)

int binarySearch(const vector& arr, int target) {
    int low = 0;
    int high = (int)arr.size() - 1;

    while (low <= high) {
        // เทคนิคป้องกัน Integer Overflow
        int mid = low + (high - low) / 2;

        if (arr[mid] == target) {
            return mid; // พบข้อมูล คืนตำแหน่ง index
        }
        else if (arr[mid] < target) {
            low = mid + 1; // ตัดครึ่งซ้ายทิ้ง ค้นหาฝั่งขวา
        }
        else {
            high = mid - 1; // ตัดครึ่งขวาทิ้ง ค้นหาฝั่งซ้าย
        }
    }
    return -1; // ตรวจสอบจนหมดช่วงแล้วไม่พบ
}

ทำไม Binary Search จึงเร็วมหาศาล?

  • ใช้หลักการ Divide and Conquer ตัดช่วงข้อมูลทิ้งรอบละ $50\%$ เสมอ
  • ข้อมูล $10^6$ ตัว (หนึ่งล้าน): ตรวจไม่เกิน 20 ครั้ง (เทียบกับ Linear Search ที่ตรวจสูงสุด $1,000,000$ ครั้ง)
  • ข้อมูล $10^9$ ตัว (หนึ่งพันล้าน): ตรวจไม่เกินเพียง 30 ครั้ง!

กับดักและข้อควรระวังในค่าย สอวน.

  • เงื่อนไขบังคับ: ข้อมูลต้องถูกเรียงลำดับ (Sorted) มาก่อนเสมอ
  • Integer Overflow: ห้ามใช้ (low + high) / 2 เพราะเสี่ยงผลบวกเกินขอบเขต int ให้ใช้ low + (high - low) / 2 เสมอ
  • ป้องกันลูปไม่รู้จบ: ต้องขยับด้วย mid + 1 หรือ mid - 1 เสมอ อย่าใส่เพียง low = mid
