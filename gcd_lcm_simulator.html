Modulo (%) Loop
int gcd(int a, int b) {
  int min = a < b ? a : b;
  for (int i = min; i >= 1; i--) {
    if (a % i == 0 && b % i == 0) {
      return i;
    }
  }
}
int lcm(int a, int b) {
  return (a / gcd(a, b)) * b; // ป้องกัน overflow
}
int gcdEuclid(int a, int b) {
  int t;
  while (b != 0) {
    t = b;
    b = a % b;
    a = t;
  }
  return a;
}
int lcm(int a, int b) {
  return (a / gcdEuclid(a, b)) * b;
}
int gcdSubtract(int a, int b) {
  while (a != b) {
    if (a > b) {
      a = a - b;
    } else {
      b = b - a;
    }
  }
  return a;
}
int lcm(int a, int b) {
  return (a / gcdSubtract(a, b)) * b;
}

แบบปกติ (Brute Force Loop)

หาค่า $\min(a, b)$ แล้วใช้ลูป for (int i = min; i >= 1; i--) ตรวจเศษ $a \% i == 0$ และ $b \% i == 0$
Time Complexity: $O(\min(a, b))$ ช้ามากเมื่อตัวเลขมีค่ามาก

แบบลบซ้ำ (Subtraction -)

ใช้ลูป while (a != b) ลบตัวมากลงด้วยตัวน้อยเรื่อยๆ จนกว่าทั้งสองจะเท่ากัน
Time Complexity: $O(\max(a, b))$ แย่ที่สุดเมื่อตัวเลขต่างกันมหาศาล เช่น $(10^6, 1)$ ลูปถึง $999,999$ รอบ

แบบหารเอาเศษ (Euclidean Modulo %)

รวบยอดการลบด้วยคำนวณ $a \pmod b$ ผ่านตัวแปรพักค่า t จนกว่า $b == 0$
Time Complexity: $O(\log \min(a, b))$ รวดเร็วที่สุด ตัวเลขระดับ $10^9$ ใช้ไม่เกิน 30 รอบ
