รอบที่: 0 / 0
ทำไมตรวจถึงแค่ √N ก็เพียงพอ? (Complementary Factor Pairs) d ≤ √N ≤ N/d

ถ้า $N = d \times d'$ หากตัวประกอบตัวหนึ่งมากกว่า $\sqrt{N}$ อีกตัวประกอบจะต้องมีค่าน้อยกว่าหรือเท่ากับ $\sqrt{N}$ เสมอ ดังนั้นหากไม่มีตัวหารใดในช่วง $[2, \lfloor\sqrt{N}\rfloor]$ ย่อมไม่มีตัวหารคู่ของมันอยู่หลัง $\sqrt{N}$ แน่นอน

bool isPrimeNaive(int n) {
  if (n <= 1) return false;
  for (int i = 2; i < n; i++) {
    if (n % i == 0) {
      return false; // พบตัวประกอบ
    }
  }
  return true; // เป็นจำนวนเฉพาะ
}
bool isPrime(int n) {
  if (n <= 1) return false;
  for (int i = 2; i * i <= n; i++) {
    if (n % i == 0) {
      return false; // ตัวหาร d <= sqrt(n)
    }
  }
  return true; // ไม่มีตัวหาร
}
bool isPrimeFast(int n) {
  if (n <= 1) return false;
  if (n <= 3) return true; // 2 และ 3
  if (n % 2 == 0 || n % 3 == 0) return false;
  for (int i = 5; i * i <= n; i += 6) {
    if (n % i == 0 || n % (i + 2) == 0)
      return false;
  }
  return true;
}

ข้อควรระวัง: i <= sqrt(n)

การเรียกฟังก์ชัน sqrt(n) ในเงื่อนไขลูปเป็นการคำนวณ Floating-point ที่ช้า และอาจเกิดความคลาดเคลื่อนทางทศนิยม ให้เปลี่ยนมาใช้ i * i <= n เสมอ

กรณี Edge Case: n ≤ 1

เลข 0, 1 และเลขติดลบ ไม่ใช่จำนวนเฉพาะ อย่าลืมใส่ if (n <= 1) return false; เพื่อป้องกันโปรแกรมคืนค่า true สำหรับเลข 1

เมื่อไหร่ใช้ Sieve แทน?

ฟังก์ชัน isPrime() เหมาะสำหรับการตรวจตัวเลขเดี่ยว $O(\sqrt{N})$ หากโจทย์ต้องการตรวจสอบจำนวนเฉพาะหลายตัวในช่วง $[1, 10^7]$ ให้เลือกใช้ Sieve of Eratosthenes $O(N \log \log N)$
