กดปุ่ม "เล่นอัตโนมัติ" หรือ "ทีละก้าว" เพื่อเริ่มดูการทำงานของ Sequential Search

sequential_search.cpp
C++ Implementation
int sequentialSearch(const vector<int>& arr, int target) {
  // วนลูปตรวจสอบสมาชิกทีละตำแหน่งจาก index 0
  for (int i = 0; i < (int)arr.size(); i++) {
    if (arr[i] == target) {
      return i; // พบข้อมูล คืนตำแหน่ง index
    }
  }
  return -1; // ตรวจครบทุกตัวแล้วไม่พบ
}
แถบสีฟ้า = บรรทัดที่กำลังทำงาน i = พอยน์เตอร์ตรวจช่อง

int sequentialSearch(const vector& arr, int target) {
    for (int i = 0; i < (int)arr.size(); i++) {
        if (arr[i] == target) {
            return i; // พบข้อมูล คืนตำแหน่ง index ทันที
        }
    }
    return -1; // ตรวจครบทุกช่องแล้วไม่พบ
}
คัดลอกโค้ดลงคลิปบอร์ดเรียบร้อยแล้ว! สามารถนำไปวาง (Ctrl+V) ใน IDE ได้ทันที

ข้อดีของ Sequential Search

  • ข้อมูลไม่ต้องเรียงลำดับ (Unsorted) ก็ค้นหาได้ทันที
  • ตรรกะเข้าใจง่าย โค้ดสั้น ไม่ซับซ้อน
  • มีประสิทธิภาพดีเยี่ยมเมื่อขนาดข้อมูล $N$ น้อย ($N \le 1,000$)

ข้อควรระวังบน Grader

  • หาก $N \ge 10^5$ และต้องค้นหาบ่อยครั้ง จะทำให้ติด Time Limit Exceeded (TLE)
  • ในค่าย สอวน. หากข้อมูลเรียงลำดับแล้ว ควรพิจารณาใช้ Binary Search ($O(\log N)$) แทน
